\begin{problem}[Uniformly accelerated motion]
A body starts from rest and moves with constant acceleration
$a=\SI{2}{\metre\per\second\squared}$. Find the distance travelled in $t=\SI{5}{\second}$.
\begin{solution}
For motion with zero initial velocity,
\[
  s = \frac{at^2}{2}
    = \frac{\SI{2}{\metre\per\second\squared}\,(\SI{5}{\second})^2}{2}
    = \SI{25}{\metre}.
\]
\end{solution}
\answer{$\SI{25}{\metre}$.}
\end{problem}
