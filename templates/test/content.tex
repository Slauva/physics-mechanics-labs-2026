\begin{problem}[Single correct answer]
Which equation represents Newton's second law?
\begin{enumerate}[label=\Alph*.]
  \item $F = ma$
  \item $p = mv$
  \item $A = Fs$
\end{enumerate}
\answer{A. $F = ma$.}
\begin{solution}
The net force on a body equals its mass multiplied by its acceleration.
\end{solution}
\end{problem}
